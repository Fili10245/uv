Esta experiencia educativa brinda técnicas para formular y resolver, mediante la computadora, problemas matemáticos cuya solución analítica es compleja o imposible. ▼ Esta semana está desarrollada por completo como ejemplo del formato.

Video: “Bisection Method”, canal Oscar Veliz.

$$ x_r = \frac{a+b}{2}, \qquad \text{con } f(a)\,f(b) < 0 $$ $$ \text{Cota del error tras } n \text{ pasos:}\quad |E_n| \le \frac{b-a}{2^{\,n+1}} $$ 
function [raiz, iter] = biseccion(f, a, b, tol, maxit)
% BISECCION  Raíz de f en [a,b] por bisección (Chapra, cap. 5).
%   f     : handle de la función, p. ej. @(x) x.^3 - x - 2
%   a, b  : extremos del intervalo, deben cumplir f(a)*f(b) < 0
%   tol   : tolerancia sobre el ancho del intervalo
%   maxit : número máximo de iteraciones
    if f(a)*f(b) > 0
        error('No hay cambio de signo en [a,b].');
    end
    iter = 0;
    while (b - a)/2 > tol && iter < maxit
        c = (a + b)/2;              % punto medio
        if f(c) == 0, break; end
        if f(a)*f(c) < 0
            b = c;                  % la raíz está en [a,c]
        else
            a = c;                  % la raíz está en [c,b]
        end
        iter = iter + 1;
    end
    raiz = (a + b)/2;
end

% --- Uso ---
f = @(x) x.^3 - x - 2;
[r, n] = biseccion(f, 1, 2, 1e-6, 100);
fprintf('Raíz ~= %.6f en %d iteraciones\n', r, n);
def biseccion(f, a, b, tol=1e-6, maxit=100):
    """Raíz de f en [a, b] por bisección (Chapra, cap. 5)."""
    if f(a) * f(b) > 0:
        raise ValueError("No hay cambio de signo en [a, b].")
    for i in range(1, maxit + 1):
        c = (a + b) / 2.0              # punto medio
        if f(c) == 0 or (b - a) / 2 < tol:
            return c, i
        if f(a) * f(c) < 0:
            b = c                     # la raíz está en [a, c]
        else:
            a = c                     # la raíz está en [c, b]
    return (a + b) / 2.0, maxit


if __name__ == "__main__":
    f = lambda x: x**3 - x - 2
    raiz, n = biseccion(f, 1, 2)
    print(f"Raíz ~= {raiz:.6f} en {n} iteraciones")
